// 1 задача
#include <iostream> // Подключение библиотеки для ввода и вывода
using namespace std; // Директива, позволяющая использовать имена из стандартной библиотеки без указания префикса

int main() { // Начало выполнения функции
    int a = 4; // присваивание значений
    int b = 5;

    a = a + b;      // a теперь равно 9 (4 + 5)
    b = a - b;      // b теперь равно 4 (9 - 5) - это старое значение a
    a = a - b;      // a теперь равно 5 (9 - 4) - это старое значение b

    cout << "a = " << a << ", b = " << b << endl;  // Вывод результата: a = 5, b = 4

    return 0; // конец выполнения функции
}

// 2 задача
#include <iostream> // Подключение библиотеки для ввода и вывода
using namespace std; // Директива, позволяющая использовать имена из стандартной библиотеки без указания префикса

int main() { // Начало выполнения функции
    int n; // Добавление переменной n
    cout << "Введите n: "; // добавление записи для более понятного ввода
    cin >> n; // ввод переменной n

    if (n < 0) { // условный оператор если (если n < 0)
        cout << "n должно быть неотрицательным." << endl; // при выполнении условия выводится текст
        return 0;
    }

    long long factorial = 1; // добавление переменной факториала
    long long result = 1; // добавление переменной результат

    for (int i = 1; i <= n; ++i) { // добавление условного оператора для (для i=1, пока i <= n, добавлять i+1)
        factorial *= i;          // факториал равен умножению самого себя на i
        result *= factorial;     // результат равен умножению самого себя на факториал
    }

    cout << "F(" << n << ") = " << result << endl; // вывод F(n) = результат
    return 0; // конец выполнения функции
}
