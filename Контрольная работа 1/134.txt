Задание 4. Вариант 12.
#include <iostream>
#include <cmath>
using namespace std;

int main() {
    // ввод коэффициентов кв. многочлена
    double a, b, c;
    cout << "Введите коэффициенты a, b, c (через пробел): ";
    cin >> a >> b >> c;

    // ввод символа
    char symbol;
    cout << "Введите символ (L, i, a): ";
    cin >> symbol;

    // константа для сравнения чисел с плавающей точкой (число близкое к 0)
    const double EPSILON = 1e-9;

    // выбор действия в зависимости от введенного символа
    switch (symbol) {
        case 'L': {
            // вывод имени и фамилии
            cout << "Тришкин Артём" << endl;
            break;
        }
        case 'i': {
            // логика нахождения корней квадратного многочлена ax^2 + bx + c
            if (abs(a) < EPSILON) {
                // уравнение не является квадратным (при a = 0)
                if (abs(b) < EPSILON) {
                    if (abs(c) < EPSILON) {
                        // a = 0, b = 0, c = 0
                        cout << "Любое действительное число x является решением (0 = 0)" << endl;
                    } else {
                        // a = 0, b = 0, c != 0
                        cout << "Решений нет (c = 0 неверно)" << endl;
                    }
                } else {
                    // линейное уравнение bx + c = 0
                    double root = -c / b;
                    cout << "Корень линейного уравнения: x = " << root << endl;
                }
            } else {
                // полное квадратное уравнение
                double discriminant = b * b - 4 * a * c;
                if (discriminant > EPSILON) {
                    // два различных действительных корня
                    double x1 = (-b + sqrt(discriminant)) / (2 * a);
                    double x2 = (-b - sqrt(discriminant)) / (2 * a);
                    cout << "Два корня: x1 = " << x1 << ", x2 = " << x2 << endl;
                } else if (abs(discriminant) < EPSILON) {
                    // один корень
                    double x = -b / (2 * a);
                    cout << "Один корень: x = " << x << endl;
                } else {
                    // действительных корней нет
                    cout << "Действительных корней нет (дискриминант < 0)" << endl;
                }
            }
            break;
        }
        case 'a': {
            // запрос дополнительного числа и проверка делимости на 11
            int number;
            cout << "Введите целое число для проверки: ";
            cin >> number;
            if (number % 11 == 0) {
                cout << "Число " << number << " делится на 11 без остатка" << endl;
            } else {
                cout << "Число " << number << " НЕ делится на 11 без остатка" << endl;
            }
            break;
        }
        default: {
            cout << "Ошибка: Введен неизвестный символ" << endl;
            break;
        }
    }

    return 0;
}
